-- ========== Roles & user-role mapping (Supabase) ==========

-- Catalog of roles
CREATE TABLE role (
  role_id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  role_name       TEXT NOT NULL UNIQUE
);

-- Map Supabase auth user <-> role
-- Supabase users table is auth.users (uuid primary key)
CREATE TABLE user_role (
  user_role_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id         UUID NOT NULL REFERENCES auth.users(id) ON UPDATE CASCADE ON DELETE CASCADE,
  role_id         BIGINT NOT NULL REFERENCES role(role_id) ON UPDATE CASCADE ON DELETE RESTRICT,
  UNIQUE (user_id, role_id)
);

-- ========== Page ==========

CREATE TABLE page (
  page_id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  page_name       TEXT NOT NULL UNIQUE
);

-- ========== Images (global) & items_images (inventory-only, per your list) ==========

CREATE TABLE images (
  image_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  image_link      TEXT NOT NULL
);

CREATE TABLE items_images (
  items_image_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  image_link      TEXT NOT NULL
);

-- ========== Internal pages & button config/setup ==========

CREATE TABLE internal_page (
  internal_page_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  internal_page_html  TEXT,
  internal_link       TEXT
);

CREATE TABLE button_config (
  button_config_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  text                  TEXT,
  sub_text              TEXT,
  text_color            TEXT,
  background_image_id   BIGINT REFERENCES images(image_id) ON UPDATE CASCADE ON DELETE SET NULL,
  background_color      TEXT,
  background_gradient   TEXT,
  link                  TEXT,
  internal_link         TEXT,
  internal_page_id      BIGINT REFERENCES internal_page(internal_page_id) ON UPDATE CASCADE ON DELETE SET NULL,
  icon                  TEXT,
  icon_color            TEXT
);

CREATE TABLE button_setup (
  button_id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  button_name       TEXT NOT NULL,
  type_id           BIGINT,
  shape_id          BIGINT,
  button_config_id  BIGINT NOT NULL REFERENCES button_config(button_config_id) ON UPDATE CASCADE ON DELETE RESTRICT,
  page_id           BIGINT NOT NULL REFERENCES page(page_id) ON UPDATE CASCADE ON DELETE CASCADE
  -- Optional if you want uniqueness per page:
  -- , UNIQUE (page_id, button_name)
);

-- ========== Events ==========

CREATE TABLE event (
  event_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  image_id        BIGINT REFERENCES images(image_id) ON UPDATE CASCADE ON DELETE SET NULL,
  title           TEXT NOT NULL,
  guest_speaker   TEXT,
  "date"          DATE,
  "time"          TIME,
  link            TEXT,
  "group"         TEXT,
  description     TEXT,
  location        TEXT,
  latitude        NUMERIC(9,6),
  longitude       NUMERIC(9,6),
  is_removed      BOOLEAN NOT NULL DEFAULT FALSE,
  CHECK (
    (latitude IS NULL AND longitude IS NULL)
    OR (latitude IS NOT NULL AND longitude IS NOT NULL) 
  )
);

-- ========== Connect Buttons & Forms ==========

CREATE TABLE forms (
  forms_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title           TEXT NOT NULL,
  link            TEXT NOT NULL
);

CREATE TABLE connect_button (
  connect_button_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  connect_button_name TEXT NOT NULL,
  forms_id            BIGINT REFERENCES forms(forms_id) ON UPDATE CASCADE ON DELETE SET NULL
);

-- ========== Slider Images ==========

CREATE TABLE slider_image (
  slider_image_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  image_id        BIGINT REFERENCES images(image_id) ON UPDATE CASCADE ON DELETE SET NULL,
  title           TEXT,
  description     TEXT,
  is_removed      BOOLEAN NOT NULL DEFAULT FALSE
);

-- ========== Inventory taxonomy ==========

CREATE TABLE item_category (
  item_category_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  item_category_name TEXT NOT NULL UNIQUE
);

CREATE TABLE item_tag (
  item_tag_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  item_tag_name   TEXT NOT NULL UNIQUE
);

-- ========== Inventory items ==========

CREATE TABLE inventory_items (
  inventory_item_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  item_name         TEXT NOT NULL,
  item_barcode      TEXT,
  item_image_id     BIGINT REFERENCES items_images(items_image_id) ON UPDATE CASCADE ON DELETE SET NULL,
  item_category_id  BIGINT REFERENCES item_category(item_category_id) ON UPDATE CASCADE ON DELETE SET NULL,
  deleted_date      TIMESTAMPTZ,
  deleted_by_id     UUID REFERENCES auth.users(id) ON UPDATE CASCADE ON DELETE SET NULL,
  created_by_id     UUID REFERENCES auth.users(id) ON UPDATE CASCADE ON DELETE SET NULL,
  created_date      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- many-to-many: inventory_items <-> item_tag
CREATE TABLE invitems_tags (
  inventory_item_id BIGINT NOT NULL REFERENCES inventory_items(inventory_item_id) ON UPDATE CASCADE ON DELETE CASCADE,
  item_tag_id       BIGINT NOT NULL REFERENCES item_tag(item_tag_id) ON UPDATE CASCADE ON DELETE CASCADE,
  PRIMARY KEY (inventory_item_id, item_tag_id)
);

-- ========== Inventory requests ==========

CREATE TABLE inventory_request (
  inventory_request_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_requesting_id   UUID REFERENCES auth.users(id) ON UPDATE CASCADE ON DELETE SET NULL,
  user_reviewing_id    UUID REFERENCES auth.users(id) ON UPDATE CASCADE ON DELETE SET NULL,
  is_approved          BOOLEAN,
  start_date           DATE,
  return_date          DATE
);

CREATE TABLE inventory_request_items (
  inventory_request_items_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  inventory_request_id       BIGINT NOT NULL REFERENCES inventory_request(inventory_request_id) ON UPDATE CASCADE ON DELETE CASCADE,
  inventory_item_id          BIGINT NOT NULL REFERENCES inventory_items(inventory_item_id) ON UPDATE CASCADE ON DELETE RESTRICT,
  is_scanned_borrowed        BOOLEAN NOT NULL DEFAULT FALSE,
  is_scanned_returned        BOOLEAN NOT NULL DEFAULT FALSE,
  return_image_id            BIGINT REFERENCES images(image_id) ON UPDATE CASCADE ON DELETE SET NULL
);

-- ========== Helpful indexes ==========

CREATE INDEX idx_button_setup_bconfig    ON button_setup(button_config_id);

CREATE INDEX idx_event_image             ON event(image_id);
CREATE INDEX idx_event_lat_lng           ON event(latitude, longitude);

CREATE INDEX idx_slider_image_image      ON slider_image(image_id);

CREATE INDEX idx_inventory_items_cat     ON inventory_items(item_category_id);
CREATE INDEX idx_inventory_items_img     ON inventory_items(item_image_id);
CREATE INDEX idx_inventory_items_created_by ON inventory_items(created_by_id);

CREATE INDEX idx_invreq_items_request    ON inventory_request_items(inventory_request_id);
CREATE INDEX idx_invreq_items_item       ON inventory_request_items(inventory_item_id);
